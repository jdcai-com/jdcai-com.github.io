Redis安装

!/bin/bash

功能：在CentOS 7.9系统安装Redis 4.0.11

安装路径：/home/admintalu/u01/tool/redis-4.0.11

注意：请使用root用户执行此脚本

检查是否为root用户

if [ "$(id -u)" -ne 0 ]; then echo "错误：请使用root用户执行此脚本" >&2 exit 1 fi

定义变量

INSTALL_PATH="/home/admintalu/u01/tool" REDIS_VERSION="4.0.11" REDIS_HOME="${INSTALL_PATH}/redis-${REDIS_VERSION}" USER="admintalu" GROUP="admintalu"

1. 安装依赖（编译工具）

echo "===== 开始安装依赖包 =====" yum install -y gcc gcc-c++ make wget &> /dev/null if [ $? -ne 0 ]; then echo "错误：依赖包安装失败" exit 1 fi

2. 准备安装目录

echo "===== 准备安装目录 =====" mkdir -p ${INSTALL_PATH} chown -R ${USER}:${GROUP} ${INSTALL_PATH} chmod 755 ${INSTALL_PATH}

3. 下载并解压Redis

echo "===== 下载并解压Redis ${REDIS_VERSION} =====" su - ${USER} -c "cd ${INSTALL_PATH} && wget -q http://download.redis.io/releases/redis-${REDIS_VERSION}.tar.gz" if [ $? -ne 0 ]; then echo "错误：Redis安装包下载失败" exit 1 fi

su - ${USER} -c "cd ${INSTALL_PATH} && tar -zxf redis-${REDIS_VERSION}.tar.gz" rm -f ${INSTALL_PATH}/redis-${REDIS_VERSION}.tar.gz # 清理安装包

4. 编译Redis

echo "===== 编译Redis =====" su - ${USER} -c "cd ${REDIS_HOME} && make" if [ $? -ne 0 ]; then echo "错误：Redis编译失败" exit 1 fi

5. 创建数据和日志目录

echo "===== 配置目录结构 =====" su - ${USER} -c "mkdir -p ${REDIS_HOME}/{data,logs,conf}" su - ${USER} -c "cp ${REDIS_HOME}/redis.conf ${REDIS_HOME}/conf/"

6. 修改Redis配置

echo "===== 配置Redis =====" su - ${USER} -c "sed -i 's|daemonize no|daemonize yes|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|pidfile /var/run/redis_6379.pid|pidfile ${REDIS_HOME}/redis.pid|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|logfile \"\"|logfile ${REDIS_HOME}/logs/redis.log|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|dir ./|dir ${REDIS_HOME}/data|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|bind 127.0.0.1|bind 0.0.0.0|g' ${REDIS_HOME}/conf/redis.conf" # 允许远程连接 su - ${USER} -c "echo 'requirepass redis@123' >> ${REDIS_HOME}/conf/redis.conf" # 设置密码 su - ${USER} -c "echo 'maxmemory 2gb' >> ${REDIS_HOME}/conf/redis.conf" # 设置最大内存 su - ${USER} -c "echo 'maxmemory-policy allkeys-lru' >> ${REDIS_HOME}/conf/redis.conf" # 内存淘汰策略

7. 配置环境变量

echo "===== 配置环境变量 =====" su - ${USER} -c "echo 'export REDIS_HOME=${REDIS_HOME}' >> /home/${USER}/.bashrc" su - ${USER} -c "echo 'export PATH=$REDIS_HOME/src:$PATH' >> /home/${USER}/.bashrc" su - ${USER} -c "source /home/${USER}/.bashrc"

8. 配置系统服务（开机自启）

echo "===== 配置系统服务 =====" cat > /etc/systemd/system/redis.service << EOF [Unit] Description=Redis Server After=network.target

[Service] Type=forking User=${USER} Group=${GROUP} ExecStart=${REDIS_HOME}/src/redis-server ${REDIS_HOME}/conf/redis.conf ExecStop=${REDIS_HOME}/src/redis-cli -a redis@123 shutdown Restart=always RestartSec=3

[Install] WantedBy=multi-user.target EOF

9. 配置防火墙

echo "===== 配置防火墙 =====" firewall-cmd --zone=public --add-port=6379/tcp --permanent &> /dev/null firewall-cmd --reload &> /dev/null

10. 启动服务并设置开机自启

echo "===== 启动Redis服务 =====" systemctl daemon-reload systemctl enable redis &> /dev/null systemctl start redis

11. 验证安装结果

echo "===== 验证安装结果 =====" sleep 3 if systemctl is-active --quiet redis; then echo "Redis启动成功！" echo "安装路径：${REDIS_HOME}" echo "状态查看：systemctl status redis" echo "客户端连接：su - ${USER} -c 'redis-cli -a redis@123'" else echo "Redis启动失败，请查看日志：${REDIS_HOME}/logs/redis.log" exit 1 fi
