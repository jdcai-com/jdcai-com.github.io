RocketMQ本地安装

!/bin/bash

功能：使用/home/admintalu/u01/tool目录下的RocketMQ安装包进行安装

前提：安装包已存在于/home/admintalu/u01/tool目录，文件名为rocketmq-all-4.9.3-bin-release.zip

安装路径：/home/admintalu/u01/tool/rocketmq-4.9.3

注意：使用root用户执行，依赖JDK 1.8+

检查root权限

if [ "$(id -u)" -ne 0 ]; then echo "错误：请使用root用户执行此脚本" >&2 exit 1 fi

定义变量

BASE_DIR="/home/admintalu/u01/tool" ROCKETMQ_VERSION="4.9.3"

安装包路径（已在BASE_DIR下）

PACKAGE_FILE="${BASE_DIR}/rocketmq-all-${ROCKETMQ_VERSION}-bin-release.zip"

解压后目录

ROCKETMQ_HOME="${BASE_DIR}/rocketmq-${ROCKETMQ_VERSION}" USER="admintalu" GROUP="admintalu" NAMESRV_PORT="9876" BROKER_PORT="10911" ZK_ADDRESS="192.168.2.18:2181" # 关联的ZooKeeper地址 STORE_PATH="${ROCKETMQ_HOME}/data/store"

1. 检查JDK

if ! java -version &> /dev/null; then echo "错误：未检测到JDK 1.8+，请先安装" exit 1 fi

2. 检查安装包是否存在

echo "===== 检查安装包 =====" if [ ! -f "${PACKAGE_FILE}" ]; then echo "错误：未在${PACKAGE_FILE}找到安装包" echo "请确保文件存在且名称正确：rocketmq-all-${ROCKETMQ_VERSION}-bin-release.zip" exit 1 fi

3. 检查目录权限

if [ "$(stat -c %U ${BASE_DIR})" != "${USER}" ]; then echo "===== 修复目录权限 =====" chown -R ${USER}:${GROUP} ${BASE_DIR} fi

4. 解压安装包（若未解压）

echo "===== 解压安装包 =====" if [ ! -d "${ROCKETMQ_HOME}" ]; then su - ${USER} -c "cd ${BASE_DIR} && unzip -q ${PACKAGE_FILE}" su - ${USER} -c "cd ${BASE_DIR} && mv rocketmq-all-${ROCKETMQ_VERSION}-bin-release rocketmq-${ROCKETMQ_VERSION}" else echo "===== 检测到已解压目录，跳过解压步骤 =====" fi

5. 创建数据存储目录

su - ${USER} -c "mkdir -p ${STORE_PATH}/commitlog ${STORE_PATH}/consumequeue ${ROCKETMQ_HOME}/logs"

6. 配置Broker

echo "===== 配置Broker =====" su - ${USER} -c "cat > ${ROCKETMQ_HOME}/conf/broker.conf << EOF brokerClusterName = DefaultCluster brokerName = broker-a brokerId = 0 deleteWhen = 04 fileReservedTime = 48 brokerRole = ASYNC_MASTER flushDiskType = ASYNC_FLUSH listenPort = ${BROKER_PORT} storePathRootDir = ${STORE_PATH} storePathCommitLog = ${STORE_PATH}/commitlog storePathConsumeQueue = ${STORE_PATH}/consumequeue namesrvAddr = localhost:${NAMESRV_PORT} zookeeperAddr = ${ZK_ADDRESS} EOF"

7. 调整JVM参数（避免内存溢出）

su - ${USER} -c "sed -i 's/-Xms4g -Xmx4g -Xmn2g/-Xms512m -Xmx512m -Xmn256m/g' ${ROCKETMQ_HOME}/bin/runserver.sh" su - ${USER} -c "sed -i 's/-Xms8g -Xmx8g -Xmn4g/-Xms512m -Xmx512m -Xmn256m/g' ${ROCKETMQ_HOME}/bin/runbroker.sh"

8. 配置系统服务

echo "===== 配置系统服务 ====="

命名服务器服务

cat > /etc/systemd/system/rocketmq-namesrv.service << EOF [Unit] Description=RocketMQ Nameserver After=network.target

[Service] Type=simple User=${USER} Group=${GROUP} ExecStart=${ROCKETMQ_HOME}/bin/mqnamesrv ExecStop=${ROCKETMQ_HOME}/bin/mqshutdown namesrv Restart=always Environment="JAVA_HOME=$(dirname $(dirname $(readlink $(readlink $(which java))))"

[Install] WantedBy=multi-user.target EOF

Broker服务

cat > /etc/systemd/system/rocketmq-broker.service << EOF [Unit] Description=RocketMQ Broker After=rocketmq-namesrv.service zookeeper.service

[Service] Type=simple User=${USER} Group=${GROUP} ExecStart=${ROCKETMQ_HOME}/bin/mqbroker -c ${ROCKETMQ_HOME}/conf/broker.conf ExecStop=${ROCKETMQ_HOME}/bin/mqshutdown broker Restart=always Environment="JAVA_HOME=$(dirname $(dirname $(readlink $(readlink $(which java))))"

[Install] WantedBy=multi-user.target EOF

9. 开放端口

firewall-cmd --zone=public --add-port=${NAMESRV_PORT}/tcp --permanent &> /dev/null firewall-cmd --zone=public --add-port=${BROKER_PORT}/tcp --permanent &> /dev/null firewall-cmd --reload &> /dev/null

10. 启动服务

echo "===== 启动RocketMQ =====" systemctl daemon-reload systemctl enable rocketmq-namesrv &> /dev/null systemctl enable rocketmq-broker &> /dev/null systemctl start rocketmq-namesrv systemctl start rocketmq-broker

11. 验证安装

sleep 5 if systemctl is-active --quiet rocketmq-namesrv && systemctl is-active --quiet rocketmq-broker; then echo "===== 安装成功 =====" echo "命名服务器地址：$(hostname -I | awk '{print $1}'):${NAMESRV_PORT}" echo "RocketMQ安装路径：${ROCKETMQ_HOME}" echo "应用配置需设置：paascloud.rocketmq.namesrvAddr=$(hostname -I | awk '{print $1}'):${NAMESRV_PORT}" else echo "===== 启动失败 =====" echo "请查看日志：${ROCKETMQ_HOME}/logs/rocketmqlogs" exit 1 fi
