Redis本地安装

!/bin/bash

功能：使用本地已上传的Redis安装包进行安装（适配指定配置）

配置信息：host=paascloud-db-redis, port=6379, password=paascloud@redis

安装路径：/home/admintalu/u01/tool/redis-4.0.11

注意：请先手动下载redis-4.0.11.tar.gz并上传到/tmp目录

请使用root用户执行此脚本

检查root权限

if [ "$(id -u)" -ne 0 ]; then echo "错误：请使用root用户执行此脚本" >&2 exit 1 fi

定义变量

LOCAL_PACKAGE="/home/admintalu/u01/tool/redis-4.0.11.tar.gz" # 本地安装包路径 INSTALL_PATH="/home/admintalu/u01/tool" REDIS_VERSION="4.0.11" REDIS_HOME="${INSTALL_PATH}/redis-${REDIS_VERSION}" USER="admintalu" GROUP="admintalu" REDIS_HOST="paascloud-db-redis" REDIS_PORT="6379" REDIS_PASSWORD="paascloud@redis"

1. 检查本地安装包是否存在

echo "===== 检查安装包 =====" if [ ! -f "${LOCAL_PACKAGE}" ]; then echo "错误：未在${LOCAL_PACKAGE}找到安装包，请先上传redis-4.0.11.tar.gz到该路径" exit 1 fi

2. 安装依赖

echo "===== 安装编译依赖 =====" yum install -y gcc gcc-c++ make &> /dev/null if [ $? -ne 0 ]; then echo "错误：依赖安装失败" exit 1 fi

3. 准备安装目录

echo "===== 准备安装目录 =====" mkdir -p ${INSTALL_PATH} chown -R ${USER}:${GROUP} ${INSTALL_PATH}

4. 复制并解压本地安装包

echo "===== 解压Redis安装包 =====" cp ${LOCAL_PACKAGE} ${INSTALL_PATH}/ su - ${USER} -c "cd ${INSTALL_PATH} && tar -zxf redis-${REDIS_VERSION}.tar.gz" rm -f ${INSTALL_PATH}/redis-${REDIS_VERSION}.tar.gz # 清理安装包

5. 编译Redis

echo "===== 编译Redis =====" su - ${USER} -c "cd ${REDIS_HOME} && make" if [ $? -ne 0 ]; then echo "错误：Redis编译失败" exit 1 fi

6. 创建必要目录

su - ${USER} -c "mkdir -p ${REDIS_HOME}/{data,logs,conf}" su - ${USER} -c "cp ${REDIS_HOME}/redis.conf ${REDIS_HOME}/conf/"

7. 配置Redis（匹配指定参数）

echo "===== 配置Redis =====" su - ${USER} -c "sed -i 's|daemonize no|daemonize yes|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|pidfile /var/run/redis_6379.pid|pidfile ${REDIS_HOME}/redis.pid|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|logfile \"\"|logfile ${REDIS_HOME}/logs/redis.log|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|dir ./|dir ${REDIS_HOME}/data|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|port 6379|port ${REDIS_PORT}|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|bind 127.0.0.1|bind 0.0.0.0|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "sed -i 's|# requirepass foobared|requirepass ${REDIS_PASSWORD}|g' ${REDIS_HOME}/conf/redis.conf" su - ${USER} -c "echo 'timeout 0' >> ${REDIS_HOME}/conf/redis.conf"

8. 配置主机名映射

echo "===== 配置主机名映射 =====" if ! grep -q "${REDIS_HOST}" /etc/hosts; then echo "$(hostname -I | awk '{print $1}') ${REDIS_HOST}" >> /etc/hosts fi

9. 配置系统服务

echo "===== 配置系统服务 =====" cat > /etc/systemd/system/redis.service << EOF [Unit] Description=Redis Server (paascloud-db-redis) After=network.target

[Service] Type=forking User=${USER} Group=${GROUP} ExecStart=${REDIS_HOME}/src/redis-server ${REDIS_HOME}/conf/redis.conf ExecStop=${REDIS_HOME}/src/redis-cli -h ${REDIS_HOST} -p ${REDIS_PORT} -a ${REDIS_PASSWORD} shutdown Restart=always

[Install] WantedBy=multi-user.target EOF

10. 配置防火墙

echo "===== 开放Redis端口 =====" firewall-cmd --zone=public --add-port=${REDIS_PORT}/tcp --permanent &> /dev/null firewall-cmd --reload &> /dev/null

11. 启动服务

echo "===== 启动Redis服务 =====" systemctl daemon-reload systemctl enable redis &> /dev/null systemctl start redis

12. 验证安装

echo "===== 验证安装结果 =====" sleep 3 if systemctl is-active --quiet redis; then echo "Redis安装成功！配置信息如下：" echo "主机名：${REDIS_HOST}" echo "端口：${REDIS_PORT}" echo "密码：${REDIS_PASSWORD}" echo "安装路径：${REDIS_HOME}" echo "测试连接：su - ${USER} -c 'redis-cli -h ${REDIS_HOST} -p ${REDIS_PORT} -a ${REDIS_PASSWORD}'" else echo "Redis启动失败，请查看日志：${REDIS_HOME}/logs/redis.log" exit 1 fi
